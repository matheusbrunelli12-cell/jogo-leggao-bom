                    tileEl.className = "tile";
                    tileEl.id = `tile-${r}-${c}`;
                    rowEl.appendChild(tileEl);
                }
                boardEl.appendChild(rowEl);
            }
        }

        function initKeyboard() {
            KEYBOARD_ROWS.forEach((row, idx) => {
                const rowEl = document.getElementById(`kbd-row-${idx + 1}`);
                rowEl.innerHTML = "";
                row.forEach(key => {
                    const btn = document.createElement("button");
                    btn.id = `key-${key}`;
                    
                    if (key === "ENTER") {
                        btn.className = "key-btn flex-[1.5] text-xs sm:text-sm";
                        btn.textContent = "ENTER";
                    } else if (key === "BACKSPACE") {
                        btn.className = "key-btn flex-[1.5]";
                        btn.innerHTML = `<svg class="w-5 h-5 pointer-events-none" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M12 14l2-2m0 0l2-2m-2 2l-2-2m2 2l2 2M3 12l6.414 6.414a2 2 0 001.414.586H19a2 2 0 002-2V7a2 2 0 002-2H10.828a2 2 0 00-1.414.586L3 12z"></path></svg>`;
                    } else {
                        btn.className = "key-btn flex-1";
                        btn.textContent = key;
                    }

                    btn.addEventListener("click", () => handleKeyPress(key));
                    rowEl.appendChild(btn);
                });
            });
        }

        function startNewGame() {
            const randomIndex = Math.floor(Math.random() * WORDS.length);
            targetWord = WORDS[randomIndex];
            
            currentRow = 0;
            currentTile = 0;
            isGameOver = false;
            boardState = Array(BOARD_ROWS).fill("").map(() => Array(WORD_LENGTH).fill(""));
            letterStatus = {};

            document.getElementById("next-game-panel").classList.add("hidden");

            for (let r = 0; r < BOARD_ROWS; r++) {
                for (let c = 0; c < WORD_LENGTH; c++) {
                    const tile = document.getElementById(`tile-${r}-${c}`);
                    tile.textContent = "";
                    tile.className = "tile";
                }
            }

            document.querySelectorAll(".key-btn").forEach(btn => {
                btn.classList.remove("correct", "present", "absent");
            });
        }

        function setupEventListeners() {
            document.addEventListener("keydown", (e) => {
                if (e.altKey || e.ctrlKey || e.metaKey) return;

                if (e.key === "Enter") {
                    handleKeyPress("ENTER");
                } else if (e.key === "Backspace") {
                    handleKeyPress("BACKSPACE");
                } else {
                    const letter = e.key.toUpperCase();
                    if (/^[A-Z]$/.test(letter) && letter.length === 1) {
                        handleKeyPress(letter);
                    }
                }
            });

            document.getElementById("btn-next").addEventListener("click", startNewGame);
            document.getElementById("btn-modal-next").addEventListener("click", () => {
                closeModal("modal-stats");
                startNewGame();
            });

            document.getElementById("btn-stats").addEventListener("click", () => {
                updateStatsUI();
                openModal("modal-stats");
            });

            document.getElementById("btn-help").addEventListener("click", () => {
                openModal("modal-help");
            });

            document.querySelectorAll(".close-modal").forEach(btn => {
                btn.addEventListener("click", (e) => {
                    const modal = e.target.closest(".fixed");
                    if (modal) closeModal(modal.id);
                });
            });
        }

        function handleKeyPress(key) {
            if (isGameOver) return;

            if (key === "ENTER") {
                submitGuess();
            } else if (key === "BACKSPACE") {
                deleteLetter();
            } else if (/^[A-Z]$/.test(key)) {
                addLetter(key);
            }
        }

        function addLetter(letter) {
            if (currentTile < WORD_LENGTH && currentRow < BOARD_ROWS) {
                boardState[currentRow][currentTile] = letter;
                const tile = document.getElementById(`tile-${currentRow}-${currentTile}`);
                tile.textContent = letter;
                tile.classList.add("pop");
                currentTile++;
            }
        }

        function deleteLetter() {
            if (currentTile > 0) {
                currentTile--;
                boardState[currentRow][currentTile] = "";
                const tile = document.getElementById(`tile-${currentRow}-${currentTile}`);
                tile.textContent = "";
                tile.classList.remove("pop");
            }
        }

        function submitGuess() {
            if (currentTile < WORD_LENGTH) {
                showToast("Palavra incompleta");
                shakeRow(currentRow);
                return;
            }

            const guess = boardState[currentRow].join("");

            // VALIDAÇÃO ESTRITA: Rejeita palpites que não sejam palavras reais do dicionário
            if (!WORDS.includes(guess)) {
                showToast("Palavra não reconhecida!");
                shakeRow(currentRow);
                return;
            }

            const targetLetters = targetWord.split("");
            const guessLetters = guess.split("");
            const rowStatuses = Array(WORD_LENGTH).fill("absent");

            // Passagem 1: Letras no lugar correto (Verde)
            for (let i = 0; i < WORD_LENGTH; i++) {
                if (guessLetters[i] === targetLetters[i]) {
                    rowStatuses[i] = "correct";
                    targetLetters[i] = null;
                }
            }

            // Passagem 2: Letras existentes em outra posição (Amarelo)
            for (let i = 0; i < WORD_LENGTH; i++) {
                if (rowStatuses[i] !== "correct") {
                    const indexInTarget = targetLetters.indexOf(guessLetters[i]);
                    if (indexInTarget !== -1) {
                        rowStatuses[i] = "present";
                        targetLetters[indexInTarget] = null;
                    }
                }
            }

            // Animação de virada sequencial
            for (let i = 0; i < WORD_LENGTH; i++) {
                const tile = document.getElementById(`tile-${currentRow}-${i}`);
                const letter = guessLetters[i];
                const status = rowStatuses[i];

                setTimeout(() => {
                    tile.classList.add("flip", status);

                    if (!letterStatus[letter] || status === "correct" || (status === "present" && letterStatus[letter] !== "correct")) {
                        letterStatus[letter] = status;
                        const keyBtn = document.getElementById(`key-${letter}`);
                        if (keyBtn) {
                            keyBtn.className = `key-btn flex-1 ${status}`;
                        }
                    }

                    if (i === WORD_LENGTH - 1) {
                        checkGameOutcome(guess);
                    }
                }, i * 200);
            }
        }

        function checkGameOutcome(guess) {
            if (guess === targetWord) {
                isGameOver = true;
                stats.played++;
                stats.wins++;
                stats.streak++;
                if (stats.streak > stats.maxStreak) stats.maxStreak = stats.streak;
                saveStats();

                setTimeout(() => {
                    showToast("Excelente! 🎉");
                    endGameUI(true);
                }, 400);

            } else if (currentRow === BOARD_ROWS - 1) {
                isGameOver = true;
                stats.played++;
                stats.streak = 0;
                saveStats();

                setTimeout(() => {
                    showToast(`A palavra era ${targetWord}`);
                    endGameUI(false);
                }, 400);

            } else {
                currentRow++;
                currentTile = 0;
            }
        }

        function endGameUI(isWin) {
            const panel = document.getElementById("next-game-panel");
            const reveal = document.getElementById("word-reveal");
            reveal.textContent = isWin ? "Você acertou!" : `A palavra era: ${targetWord}`;
            panel.classList.remove("hidden");

            const modalNextBtn = document.getElementById("btn-modal-next");
            modalNextBtn.classList.remove("hidden");

            setTimeout(() => {
                updateStatsUI();
                openModal("modal-stats");
            }, 1000);
        }

        function shakeRow(rowIdx) {
            const rowEl = document.getElementById(`row-${rowIdx}`);
            rowEl.classList.add("row-shake");
            setTimeout(() => {
                rowEl.classList.remove("row-shake");
            }, 400);
        }

        function showToast(msg) {
            const toast = document.getElementById("toast");
            toast.textContent = msg;
            toast.classList.remove("opacity-0", "-translate-y-2", "pointer-events-none");
            toast.classList.add("opacity-100", "translate-y-0");

            setTimeout(() => {
                toast.classList.remove("opacity-100", "translate-y-0");
                toast.classList.add("opacity-0", "-translate-y-2", "pointer-events-none");
            }, 2000);
        }

        function saveStats() {
            localStorage.setItem("letreco_infinito_stats", JSON.stringify(stats));
        }

        function updateStatsUI() {
            document.getElementById("stat-played").textContent = stats.played;
            const winrate = stats.played > 0 ? Math.round((stats.wins / stats.played) * 100) : 0;
            document.getElementById("stat-winrate").textContent = `${winrate}%`;
            document.getElementById("stat-streak").textContent = stats.streak;
            document.getElementById("stat-maxstreak").textContent = stats.maxStreak;
        }

        function openModal(modalId) {
            const modal = document.getElementById(modalId);
            modal.classList.remove("hidden");
            setTimeout(() => {
                modal.classList.remove("opacity-0");
            }, 10);
        }

        function closeModal(modalId) {
            const modal = document.getElementById(modalId);
            modal.classList.add("opacity-0");
            setTimeout(() => {
                modal.classList.add("hidden");
            }, 200);
        }
